cmake_minimum_required(VERSION 3.21)
project(Lab09Academico LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_library(academico src/model.cpp src/controller.cpp src/controller.h)
target_include_directories(academico PUBLIC src)
target_link_libraries(academico PUBLIC Qt6::Core)
add_executable(lab09 src/main.cpp src/mainwindow.cpp src/mainwindow.h)
target_link_libraries(lab09 PRIVATE academico Qt6::Widgets)
enable_testing()
add_executable(pruebas tests/test_academico.cpp)
target_link_libraries(pruebas PRIVATE academico Qt6::Test)
add_test(NAME modelo_controlador COMMAND pruebas)
add_library(baseline OBJECT EXCLUDE_FROM_ALL comparacion/antes.cpp)
target_link_libraries(baseline PRIVATE Qt6::Core)
